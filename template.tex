% === CODE BLOCK STYLING ===

% 1. Define the visual style of the code block
\usepackage{tcolorbox}
\tcbuselibrary{breakable}

% Define the colors (RGB format)
\definecolor{codebackground}{RGB}{242,242,242} % Light Gray
\definecolor{codeframe}{RGB}{220,220,220}      % Slightly darker border

% 2. Apply style to syntax-highlighted code blocks (e.g., ```bash)
\makeatletter
\@ifundefined{Shaded}{}{
  \renewenvironment{Shaded}{
    \begin{tcolorbox}[
      breakable,
      colback=codebackground,
      colframe=codeframe,
      boxrule=0.5pt,
      arc=3pt,
      left=4pt, right=4pt, top=4pt, bottom=4pt
    ]
  }{
    \end{tcolorbox}
  }
}
\makeatother

% 3. Apply style to plain/unhighlighted code blocks (e.g., ```)
% NOTE: verbatim must not be redefined with \renewenvironment (breaks catcode
% scanning). Hooks preserve the original verbatim behaviour.
\usepackage{etoolbox}
\BeforeBeginEnvironment{verbatim}{%
  \begin{tcolorbox}[
    breakable,
    colback=codebackground,
    colframe=codeframe,
    boxrule=0.5pt,
    arc=3pt,
    left=4pt, right=4pt, top=4pt, bottom=4pt
  ]%
}
\AfterEndEnvironment{verbatim}{%
  \end{tcolorbox}%
}
